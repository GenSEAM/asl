fn runLengthEncode text: String -> (List (Pair String Int64))
  (rleFrom text 0 (stringLength text) (list))

fn rleFrom text: String i: Int64 n: Int64 acc: (List (Pair String Int64)) -> (List (Pair String Int64))
  (cond
    ((>= i n) acc)
    (else
      (match (stringSlice text i (+ i 1))
        ((some c)
          (let [(j (runEnd text (+ i 1) n c))]
            (rleFrom text j n (listAppend acc (list (pair c (- j i)))))))
        ((none) acc))))

fn runEnd text: String i: Int64 n: Int64 c: String -> Int64
  (cond
    ((>= i n) n)
    (else
      (match (stringSlice text i (+ i 1))
        ((some ch) (if (stringEq? ch c) (runEnd text (+ i 1) n c) i))
        ((none) n))))
