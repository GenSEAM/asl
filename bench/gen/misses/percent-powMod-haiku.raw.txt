fn powModGo b: Int64 e: Int64 m: Int64 acc: Int64 -> Int64
  (cond
      ((= e 0) acc)
      ((= (% e 2) 1) (powModGo (% (* b b) m) (/ e 2) m (% (* acc b) m)))
      (else (powModGo (% (* b b) m) (/ e 2) m acc)))

fn powMod base: Int64 exponent: Int64 modulus: Int64 -> Int64
  (powModGo (% base modulus) exponent modulus (% 1 modulus))
