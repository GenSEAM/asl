fn mergeSorted a: (List Int64) b: (List Int64) -> (List Int64)
  (match (listHead a)
    ((none) b)
    ((some x)
      (match (listHead b)
        ((none) a)
        ((some y)
          (cond
            ((<= x y)
              (match (listTail a)
                ((some ta) (listCons x (mergeSorted ta b)))
                ((none) (listCons x b))))
            (else
              (match (listTail b)
                ((some tb) (listCons y (mergeSorted a tb)))
                ((none) (listCons y a)))))))))
