(:card :lang ASL :reply "code at once")
(:rule :id body :do "bare fn header line, then ONE parenthesised form under it" :ex "fn double n: Int64 -> Int64\n  (* n 2)" :never "bare cond, match or let as the body")
(:rule :id call :do "close each form once" :ex (+ a (* b 2)))
(:rule :id equal :do "= != mod; / divides Int64 whole" :ex (= (mod n 2) 0) :never (== %))
(:rule :id branch :ex ((if c x y) (cond ((< n 0) "neg") (else "pos"))))
(:rule :id list :do "listAppend joins two lists; listDrop xs 1 is the rest" :ex ((list 1 2) (list) (listAppend acc (list x))) :never ("[]" len head))
(:rule :id option :do "listHead listGet listMax mapGet stringToInt64 return (some v) or (none)" :ex ((match (listHead xs) ((none) 0) ((some h) h)) (optionOr (mapGet m k) 0)))
(:rule :id names :do "camelCase, type first" :have (listLength listReverse listSort listContains? listSum stringLength stringChars stringSplit stringJoin))
(:rule :id string :do "str joins, + is numeric" :ex (str "n=" (stringFromInt64 n)))
(:rule :id bind :ex ((let [(a 1) (b (+ a 1))] (* a b)) (fold (fn [acc x] (+ acc x)) 0 xs)))
(:rule :id enum :do "constructors are calls: (circle 3) (dot)" :ex (match s ((circle r) (* r r)) ((dot) 0)))
(:rule :id record :ex (user.age (with user :age 1)))
(:rule :id result :ex ((ok v) (err "message")))
